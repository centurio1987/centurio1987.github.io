export const TreeNode = ({ node }) => {
  const isUser = node.source === 'USER' || node.source === 'USER_EXPLICIT';
  const isModel = node.source === 'MODEL';
  const nodeClass = `tree-node ${isUser ? 'tree-node-user' : isModel ? 'tree-node-model' : 'tree-node-system'}`;
  
  return (
    <li>
      <a href={`#${node.id}`} className={nodeClass}>
        <strong>{node.source}</strong><br/>
        <span style={{color: '#888', fontSize: '11px'}}>{node.type}</span><br/>
        <span style={{color: '#bbb', fontSize: '10px'}}>Step {node.step_index}</span>
      </a>
      {node.children && node.children.length > 0 && (
        <ul>
          {node.children.map(child => <TreeNode key={child.id} node={child} />)}
        </ul>
      )}
    </li>
  );
};

export const ConversationTree = ({ roots }) => {
  return (
    <div className="tree-container">
      <div className="tree">
        <ul>
          {roots.map(root => <TreeNode key={root.id} node={root} />)}
        </ul>
      </div>
    </div>
  );
};

export const ConversationDetails = ({ nodes }) => {
  // 중복 렌더링 방지를 위해 id 기준으로 필터링(children 등에 의해 중첩된 배열일 경우를 대비)
  const uniqueNodes = Array.from(new Map(nodes.map(n => [n.id, n])).values());
  
  return (
    <div style={{ marginTop: '40px', fontFamily: 'sans-serif' }}>
      <h2>대화 상세 내역</h2>
      {uniqueNodes.map(node => (
        <div key={node.id} id={node.id} style={{
          padding: "20px", 
          border: "1px solid #eee", 
          borderRadius: "8px", 
          marginBottom: "20px",
          background: node.source === 'USER' ? '#fafffa' : node.source === 'MODEL' ? '#f4f9ff' : '#fffaf4'
        }}>
          <h3 style={{ margin: '0 0 15px 0', borderBottom: '1px solid #ddd', paddingBottom: '10px' }}>
            Step {node.step_index} : {node.source} <span style={{fontSize: '14px', color: '#666', fontWeight: 'normal'}}>({node.type})</span>
          </h3>
          {node.content && (
            <div style={{
              whiteSpace: 'pre-wrap', 
              wordWrap: 'break-word', 
              background: '#fff', 
              border: '1px solid #eee',
              padding: '15px', 
              borderRadius: '5px',
              fontSize: '14px',
              lineHeight: '1.6',
              color: '#333'
            }}>
              {node.content}
            </div>
          )}
          {node.tool_calls && node.tool_calls.length > 0 && (
            <div style={{ marginTop: '15px' }}>
              <strong>Tool Calls:</strong>
              <pre style={{
                background: '#2d2d2d', 
                color: '#ccc', 
                padding: '15px', 
                borderRadius: '5px',
                overflowX: 'auto',
                fontSize: '13px'
              }}>
                {JSON.stringify(node.tool_calls, null, 2)}
              </pre>
            </div>
          )}
        </div>
      ))}
    </div>
  );
};
